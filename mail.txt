//Obtenemos datos básicos del usuario por PkUserweb.
async function getUserInfo(pkUserweb) {
  const rows = await UbHrDataSource.manager.query(
    `SELECT
       PkUserweb,
       username,
       Nombre,
       Apellidos
     FROM UbUsersweb
     WHERE PkUserweb = @0`,
    [pkUserweb]
  );

  if (!rows || rows.length === 0) return null;
  return rows[0];
}

//Obtenemos datos básicos del usuario por username (campo 'usuario' en compras_extraordinarias).
async function getUserInfoByUsername(username) {
  const rows = await UbHrDataSource.manager.query(
    `SELECT
       PkUserweb,
       username,
       Nombre,
       Apellidos
     FROM UbUsersweb
     WHERE username = @0`,
    [username]
  );

  if (!rows || rows.length === 0) return null;
  return rows[0];
}

//Envía correo a los directores cuando se crea una nueva solicitudy copia al solicitante (a partir de username).
async function enviarCorreoNuevaSolicitud({ pkUserweb, titulo, coste, fecha, proyecto }) {
  console.log('🟦 [enviarCorreoNuevaSolicitud] Inicio', {
    pkUserweb,
    titulo,
    coste,
    fecha,
    proyecto
  });

  const user = await getUserInfo(pkUserweb);

  if (!user) {
    console.warn('🟨 Usuario no encontrado para enviar correo. pkUserweb =', pkUserweb);
    return;
  }

  const responsables = resolveDestinatarios(user);
  if (!responsables || responsables.length === 0) {
    console.warn('🟨 Sin responsables configurados para usuario:', user.username);
    return;
  }

  const solicitanteNombre =
    `${user.Nombre || ''} ${user.Apellidos || ''}`.trim() || user.username;

  const correoSolicitante = user.username.includes('@')
    ? user.username
    : `${user.username}@grupoub.com`;

  // Montamos el "to" con todos los responsables
  const toField = responsables
    .map(r => `${r.nombre} <${r.email}>`)
    .join(', ');

  const mailOptions = {
    from: '"Notificaciones de mis estados de compra" <cau@grupoub.com>',
    to: toField,
    cc: correoSolicitante, // copia al solicitante
    subject: `Nueva Solicitud para Aprobar - ${titulo}`,
    text: `
Hola,

Se ha registrado una nueva solicitud de compra extraordinaria.

Solicitante: ${solicitanteNombre} (${user.username})
Correo solicitante: ${correoSolicitante}

Título: ${titulo}
Proyecto: ${proyecto || '-'}
Fecha: ${fecha}
Coste: ${coste} €

Por favor, revisad la solicitud en la aplicación.

Saludos,
Equipo de Compras
`
  };

  console.log('🟦 [enviarCorreoNuevaSolicitud] Opciones de correo construidas:', {
    to: mailOptions.to,
    cc: mailOptions.cc,
    subject: mailOptions.subject
  });

  try {
    const info = await transporter.sendMail(mailOptions);
    console.log('🟩 [enviarCorreoNuevaSolicitud] Correo enviado OK. messageId:', info.messageId);
  } catch (err) {
    console.error('🟥 [enviarCorreoNuevaSolicitud] Error al enviar correo:', err);
    throw err;
  }
}
